import { Tensor, add, matmul, gather, relu, rmsnorm, attention } from './tensor.js';

/** Tokenizer lexical volontairement fermé, distinct des BPE de production. */
export function lexicalPieces(text) {
  return text.normalize('NFC').toLocaleLowerCase('fr').match(/[\p{L}\p{N}]+|[^\s\p{L}\p{N}]/gu) || [];
}

export function tokenize(text, vocabulary) {
  const words = lexicalPieces(text);
  const ids = words.map(word => vocabulary.indexOf(word));
  const unknown = words.filter((_, i) => ids[i] < 0);
  if (unknown.length) { throw new Error(`Hors vocabulaire : ${[...new Set(unknown)].join(', ')}. Utilisez un exemple ou les mots du corpus.`); }
  return [vocabulary.indexOf('<bos>'), ...ids];
}

/** Vérifie les poids avant toute inférence ; aucun remplacement implicite. */
export function loadWeights(model, train = false) {
  const c = model.config;
  if (!c || !Number.isInteger(c.dim) || c.dim < 2 || c.dim > 128 ||
    !Number.isInteger(c.layers) || c.layers < 1 || c.layers > 8 ||
    !Number.isInteger(c.heads) || c.heads < 1 || c.dim % c.heads ||
    !Number.isInteger(c.context) || c.context < 2 || c.context > 128 ||
    !Array.isArray(model.vocabulary) || model.vocabulary.length < 3 ||
    !model.vocabulary.includes('<bos>') || !model.vocabulary.includes('<eos>')) {
    throw new Error('Configuration du mini-modèle invalide.');
  }
  const shapes = { token: [model.vocabulary.length, c.dim], position: [c.context, c.dim], output: [c.dim, model.vocabulary.length] };
  for (let i = 0; i < c.layers; i++) {
    for (const name of ['q', 'k', 'v', 'o']) { shapes[`${i}.${name}`] = [c.dim, c.dim]; }
    shapes[`${i}.up`] = [c.dim, c.dim * 2];
    shapes[`${i}.down`] = [c.dim * 2, c.dim];
  }
  return Object.fromEntries(Object.entries(shapes).map(([name, [rows, cols]]) => {
    const data = model.weights?.[name];
    if (!Array.isArray(data) || data.length !== rows * cols || data.some(v => !Number.isFinite(v))) {
      throw new Error(`Poids invalides : ${name}.`);
    }
    return [name, new Tensor(data, rows, cols, train)];
  }));
}

/** Inférence exacte : chaque trace correspond aux tenseurs utilisés pour prédire. */
export function forward(ids, weights, config) {
  if (!ids.length || ids.length > config.context) {
    throw new RangeError(`Le mini-modèle accepte de 1 à ${config.context} tokens, début compris.`);
  }
  const embedding = gather(weights.token, ids);
  let x = add(embedding, gather(weights.position, ids.map((_, i) => i)));
  const traces = [{ state: x, embedding }];
  for (let i = 0; i < config.layers; i++) {
    const normalized = rmsnorm(x);
    const q = matmul(normalized, weights[`${i}.q`]);
    const k = matmul(normalized, weights[`${i}.k`]);
    const v = matmul(normalized, weights[`${i}.v`]);
    const attended = attention(q, k, v, config.heads);
    const residual = add(x, matmul(attended.result, weights[`${i}.o`]));
    const hidden = relu(matmul(rmsnorm(residual), weights[`${i}.up`]));
    x = add(residual, matmul(hidden, weights[`${i}.down`]));
    traces.push({ state: x, attention: attended.maps, q, k, v, hidden });
  }
  return { logits: matmul(rmsnorm(x), weights.output), traces };
}

export function logitLens(state, weights) {
  return matmul(rmsnorm(state), weights.output).toRows().at(-1);
}

export function decode(ids, vocabulary) {
  return ids.map(id => vocabulary[id]).filter(t => t !== '<bos>' && t !== '<eos>')
    .join(' ').replace(/\s+([.,!?;:])/g, '$1');
}

/**
 * Petit moteur matriciel différentiable. Les gradients ne sont alloués que
 * pendant l'entraînement ; l'inférence utilise exactement les mêmes opérations.
 */
export class Tensor {
  constructor(data, rows, cols, train = false) {
    if (data.length !== rows * cols || rows < 1 || cols < 1) {
      throw new RangeError('Dimensions de tenseur incompatibles.');
    }
    this.data = Float64Array.from(data);
    this.rows = rows;
    this.cols = cols;
    this.grad = train ? new Float64Array(data.length) : null;
    this.parents = [];
    this.backwardOp = () => {};
  }

  /** Propage un gradient scalaire une fois dans l'ordre topologique inverse. */
  backward() {
    if (!this.grad || this.data.length !== 1) {
      throw new Error('La rétropropagation exige une perte scalaire différentiable.');
    }
    const visited = new Set();
    const order = [];
    const visit = tensor => {
      if (visited.has(tensor)) { return; }
      visited.add(tensor);
      tensor.parents.forEach(visit);
      order.push(tensor);
    };
    visit(this);
    this.grad[0] = 1;
    order.reverse().forEach(tensor => tensor.backwardOp());
  }

  toRows() {
    return Array.from({ length: this.rows }, (_, i) =>
      Array.from(this.data.slice(i * this.cols, (i + 1) * this.cols)));
  }
}

function operation(data, rows, cols, parents, backward) {
  const result = new Tensor(data, rows, cols, parents.some(p => p.grad));
  if (result.grad) {
    result.parents = parents;
    result.backwardOp = () => backward(result.grad);
  }
  return result;
}

export function add(a, b) {
  if (a.rows !== b.rows || a.cols !== b.cols) { throw new RangeError('Addition incompatible.'); }
  return operation(a.data.map((v, i) => v + b.data[i]), a.rows, a.cols, [a, b], grad => {
    grad.forEach((v, i) => {
      if (a.grad) { a.grad[i] += v; }
      if (b.grad) { b.grad[i] += v; }
    });
  });
}

export function matmul(a, b) {
  if (a.cols !== b.rows) { throw new RangeError('Produit matriciel incompatible.'); }
  const data = new Float64Array(a.rows * b.cols);
  for (let i = 0; i < a.rows; i++) {
    for (let k = 0; k < a.cols; k++) {
      for (let j = 0; j < b.cols; j++) {
        data[i * b.cols + j] += a.data[i * a.cols + k] * b.data[k * b.cols + j];
      }
    }
  }
  return operation(data, a.rows, b.cols, [a, b], grad => {
    for (let i = 0; i < a.rows; i++) {
      for (let k = 0; k < a.cols; k++) {
        for (let j = 0; j < b.cols; j++) {
          const g = grad[i * b.cols + j];
          if (a.grad) { a.grad[i * a.cols + k] += g * b.data[k * b.cols + j]; }
          if (b.grad) { b.grad[k * b.cols + j] += g * a.data[i * a.cols + k]; }
        }
      }
    }
  });
}

export function gather(table, ids) {
  if (!ids.length || ids.some(id => !Number.isInteger(id) || id < 0 || id >= table.rows)) {
    throw new RangeError('Identifiant hors du vocabulaire.');
  }
  const data = ids.flatMap(id => Array.from(table.data.slice(id * table.cols, (id + 1) * table.cols)));
  return operation(data, ids.length, table.cols, [table], grad => {
    if (!table.grad) { return; }
    ids.forEach((id, i) => {
      for (let j = 0; j < table.cols; j++) { table.grad[id * table.cols + j] += grad[i * table.cols + j]; }
    });
  });
}

export function relu(a) {
  return operation(a.data.map(v => Math.max(0, v)), a.rows, a.cols, [a], grad => {
    if (a.grad) { grad.forEach((v, i) => { a.grad[i] += a.data[i] > 0 ? v : 0; }); }
  });
}

/** RMSNorm sans gain appris : x / sqrt(moyenne(x²) + epsilon). */
export function rmsnorm(a) {
  const data = new Float64Array(a.data.length);
  const scales = [];
  for (let i = 0; i < a.rows; i++) {
    const row = a.data.slice(i * a.cols, (i + 1) * a.cols);
    const scale = 1 / Math.sqrt(row.reduce((s, v) => s + v * v, 0) / a.cols + 1e-5);
    scales.push(scale);
    row.forEach((v, j) => { data[i * a.cols + j] = v * scale; });
  }
  return operation(data, a.rows, a.cols, [a], grad => {
    if (!a.grad) { return; }
    for (let i = 0; i < a.rows; i++) {
      let dot = 0;
      for (let j = 0; j < a.cols; j++) { dot += grad[i * a.cols + j] * a.data[i * a.cols + j]; }
      for (let j = 0; j < a.cols; j++) {
        const k = i * a.cols + j;
        a.grad[k] += grad[k] * scales[i] - a.data[k] * scales[i] ** 3 * dot / a.cols;
      }
    }
  });
}

/** Attention multi-tête causale, avec cartes exactes pour l'inspection. */
export function attention(q, k, v, heads) {
  if (q.cols % heads || q.rows !== k.rows || q.rows !== v.rows || q.cols !== k.cols || q.cols !== v.cols) {
    throw new RangeError('Dimensions des têtes incompatibles.');
  }
  const width = q.cols / heads;
  const n = q.rows;
  const scale = 1 / Math.sqrt(width);
  const maps = Array.from({ length: heads }, () => Array.from({ length: n }, () => Array(n).fill(0)));
  const data = new Float64Array(n * q.cols);
  for (let h = 0; h < heads; h++) {
    for (let i = 0; i < n; i++) {
      const scores = [];
      for (let j = 0; j <= i; j++) {
        let score = 0;
        for (let d = 0; d < width; d++) { score += q.data[i * q.cols + h * width + d] * k.data[j * q.cols + h * width + d]; }
        scores.push(score * scale);
      }
      const max = Math.max(...scores);
      const exps = scores.map(s => Math.exp(s - max));
      const sum = exps.reduce((s, x) => s + x, 0);
      exps.forEach((x, j) => { maps[h][i][j] = x / sum; });
      for (let d = 0; d < width; d++) {
        for (let j = 0; j <= i; j++) { data[i * q.cols + h * width + d] += maps[h][i][j] * v.data[j * q.cols + h * width + d]; }
      }
    }
  }
  const result = operation(data, n, q.cols, [q, k, v], grad => {
    for (let h = 0; h < heads; h++) {
      for (let i = 0; i < n; i++) {
        const dp = Array(i + 1).fill(0);
        for (let j = 0; j <= i; j++) {
          for (let d = 0; d < width; d++) {
            const qi = i * q.cols + h * width + d;
            const kj = j * q.cols + h * width + d;
            dp[j] += grad[qi] * v.data[kj];
            if (v.grad) { v.grad[kj] += maps[h][i][j] * grad[qi]; }
          }
        }
        const mean = dp.reduce((s, x, j) => s + x * maps[h][i][j], 0);
        for (let j = 0; j <= i; j++) {
          const ds = maps[h][i][j] * (dp[j] - mean) * scale;
          for (let d = 0; d < width; d++) {
            const qi = i * q.cols + h * width + d;
            const kj = j * q.cols + h * width + d;
            if (q.grad) { q.grad[qi] += ds * k.data[kj]; }
            if (k.grad) { k.grad[kj] += ds * q.data[qi]; }
          }
        }
      }
    }
  });
  return { result, maps };
}

export function crossEntropy(logits, targets) {
  if (targets.length !== logits.rows || targets.some(t => !Number.isInteger(t) || t < 0 || t >= logits.cols)) {
    throw new RangeError('Cibles incompatibles.');
  }
  let loss = 0;
  const probabilities = logits.toRows().map((row, i) => {
    const max = Math.max(...row);
    const exp = row.map(v => Math.exp(v - max));
    const sum = exp.reduce((s, v) => s + v, 0);
    loss += Math.log(sum) + max - row[targets[i]];
    return exp.map(v => v / sum);
  });
  return operation([loss / targets.length], 1, 1, [logits], grad => {
    if (!logits.grad) { return; }
    probabilities.forEach((row, i) => row.forEach((p, j) => {
      logits.grad[i * logits.cols + j] += (p - Number(j === targets[i])) / targets.length * grad[0];
    }));
  });
}

import { SeededRandom } from '../../../../lib/seeded-random.js';

export function softmax(logits, temperature = 1) {
  if (!logits.length || logits.some(v => !Number.isFinite(v)) || !Number.isFinite(temperature) || temperature < 0) {
    throw new RangeError('Logits finis et température positive ou nulle requis.');
  }
  const max = Math.max(...logits);
  if (temperature === 0) { return logits.map((_, i) => Number(i === logits.indexOf(max))); }
  const exp = logits.map(v => Math.exp((v - max) / temperature));
  const sum = exp.reduce((s, v) => s + v, 0);
  return exp.map(v => v / sum);
}

/** Ordre explicite : température, top-k, min-p, top-p, puis renormalisation. */
export function distribution(logits, temperature = 1, topP = 1, minP = 0, topK = logits.length) {
  if (!Number.isFinite(topP) || topP <= 0 || topP > 1 || !Number.isFinite(minP) || minP < 0 || minP > 1) {
    throw new RangeError('top-p doit être dans ]0,1] et min-p dans [0,1].');
  }
  const probabilities = softmax(logits, temperature);
  if (!Number.isInteger(topK) || topK < 1 || topK > logits.length) {
    throw new RangeError('top-k doit être un entier entre 1 et la taille du vocabulaire.');
  }
  const peak = Math.max(...probabilities);
  const eligible = probabilities.map((p, id) => ({ p, id }))
    .filter(({ p }) => p > 0).sort((a, b) => b.p - a.p || a.id - b.id)
    .slice(0, topK).filter(({ p }) => p >= peak * minP);
  const total = eligible.reduce((s, item) => s + item.p, 0);
  const kept = Array(logits.length).fill(0);
  let cumulative = 0;
  for (const { p, id } of eligible) {
    kept[id] = p;
    cumulative += p;
    if (cumulative >= topP * total) { break; }
  }
  return kept.map(p => p / cumulative);
}

export function sample(probabilities, rng) {
  if (!probabilities.length || probabilities.some(p => !Number.isFinite(p) || p < 0) ||
    Math.abs(probabilities.reduce((s, p) => s + p, 0) - 1) > 1e-8) {
    throw new RangeError('Distribution normalisée requise.');
  }
  const draw = rng.random();
  let sum = 0;
  for (let i = 0; i < probabilities.length; i++) {
    sum += probabilities[i];
    if (draw < sum) { return i; }
  }
  return probabilities.findLastIndex(p => p > 0);
}

export function histogram(probabilities, count = 100, seed = 42) {
  if (!Number.isInteger(count) || count < 1 || count > 10000) { throw new RangeError('Nombre de tirages invalide.'); }
  const rng = new SeededRandom(seed);
  const bins = probabilities.map(() => 0);
  for (let i = 0; i < count; i++) { bins[sample(probabilities, rng)]++; }
  return bins;
}

export function kvBytes({ layers, kvHeads, headDim, tokens, batch, bytes }) {
  const values = [layers, kvHeads, headDim, tokens, batch, bytes];
  if (values.some(v => !Number.isFinite(v) || v <= 0)) { throw new RangeError('Dimensions du cache strictement positives requises.'); }
  return 2 * layers * kvHeads * headDim * tokens * batch * bytes;
}

export function roofline({ parameters, batch, weightBytes, bandwidth, flops }) {
  if ([parameters, batch, weightBytes, bandwidth, flops].some(v => !Number.isFinite(v) || v <= 0)) {
    throw new RangeError('Paramètres matériels strictement positifs requis.');
  }
  const operations = 2 * parameters * batch;
  const readBytes = parameters * weightBytes;
  const computeSeconds = operations / flops;
  const memorySeconds = readBytes / bandwidth;
  return { intensity: operations / readBytes, computeSeconds, memorySeconds, seconds: Math.max(computeSeconds, memorySeconds) };
}

export function rotatePair(x, y, position, frequency = 1) {
  const angle = position * frequency;
  return [x * Math.cos(angle) - y * Math.sin(angle), x * Math.sin(angle) + y * Math.cos(angle)];
}

/** Rotation orthogonale pour l'atlas ; aucune modification des embeddings. */
export function rotate3D(point, azimuth, elevation) {
  if (point.length !== 3 || [...point, azimuth, elevation].some(v => !Number.isFinite(v))) {
    throw new RangeError('Trois coordonnées et deux angles finis sont requis.');
  }
  const [x, y, z] = point;
  const horizontal = x * Math.cos(azimuth) + z * Math.sin(azimuth);
  const depth = -x * Math.sin(azimuth) + z * Math.cos(azimuth);
  return [horizontal, y * Math.cos(elevation) - depth * Math.sin(elevation),
    y * Math.sin(elevation) + depth * Math.cos(elevation)];
}
